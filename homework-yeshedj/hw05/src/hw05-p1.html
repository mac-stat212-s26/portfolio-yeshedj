library(tidyverse)

summarize_toplevels <- function(f, n) {
  lvls <- levels(f)
  top_lvls<- lvls[seq_len(n)]
  bot_lvls<- lvls[seq(length(lvls) - n + 1, length(lvls))]
  mid_lvls<- setdiff(lvls, c(top_lvls, bot_lvls))

  groups <-list(top_lvls, bot_lvls) 
  names(groups) <- c(paste(top_lvls, collapse = ", "), 
                     paste(bot_lvls, collapse = ", "))
  
  if (length(mid_lvls) > 0)
    groups[[paste(mid_lvls, collapse = ", ")]] <- mid_lvls

  f_grouped <- fct_collapse(f, !!!groups)

  tibble(f = f_grouped) |>
    count(f, .drop = FALSE) |>
    mutate(percent = n / sum(n))
}

summarize_toplevels(f, n = 2)

clean_names2 <- function(data) {
  # Your code here, using 'data' as the name of the dataset
  cnames <- names(data)

  cnames <- str_replace_all(cnames, "%", "percent")
  cnames <- str_replace_all(cnames, "#", "number")
  cnames <- str_replace_all(cnames, "[^a-zA-Z0-9]+", "_")
  cnames <- str_to_lower(cnames)
  cnames <- str_remove_all(cnames,"^_|_$")

  for (d in unique(cnames[duplicated(cnames)])) {
    index <- which(cnames == d)
    cnames[index] <- paste0(d, seq_along(index))
  }

  names(data) <- cnames
  data
}

Exercise 2: recycling_checker()

  • Inputs:
    • x1 and x2: two numeric vectors
  • Function purpose: Print a helpful message to the user warning them if dangerous vector recycling will happen when they try to combine the two vectors with some arithmetic operation.
    • If the vectors are not of the same length, provide informational text. See examples below for how to format this text.
      • Example 1: x1 is length 4 and x2 is length 2. Informational text: “x1 and x2 are not the same length. To match the length of x1, x2 will be repeated 2 times. You will not get a warning when you combine these two vectors, but just be wary that vector recycling is occurring.”
      • Example 2: x1 is length 3 and x2 is length 11. Informational text: “x1 and x2 are not the same length. To match the length of x2, x1 will be repeated 3 times fully with the first 2 elements repeated again. You will get a warning when you combine these two vectors.”
Code
# Write your function here

recycling_checker <- function(x1, x2) {
  len1 <- length(x1)
  len2 <- length(x2)

  if (len1 == len2) {
    message("x1 and x2 are the same length. No recycling will occur.")
    return(invisible(NULL))
  }

  if (len1 > len2) {
    longer <- "x1"; shorter <- "x2"
    len_larger <- len1; len_smaller <- len2
  } else {
    longer <- "x2"; shorter <-"x1"
    len_larger <- len2; len_smaller <- len1
  }

  times_full <- len_larger %/% len_smaller
  remainder  <- len_larger %%  len_smaller

  if (remainder == 0) {
    message(str_glue(
      "x1 and x2 are not the same length. To match the length of {longer}, ",
      "{shorter} will be repeated {times_full} times. ",
      "You will not get a warning when you combine these two vectors, ",
      "but just be wary that vector recycling is occurring."
    ))
  } else {
    message(str_glue(
      "x1 and x2 are not the same length. To match the length of {longer}, ",
      "{shorter} will be repeated {times_full} times fully with the first ",
      "{remainder} element{if (remainder == 1) '' else 's'} repeated again. ",
      "You will get a warning when you combine these two vectors."
    ))
  }
}

Exercise 3: days_to_birthday()

  • Function purpose: Display the number of days from a user-supplied day to the user’s next birthday. For example, if the user enters the day “Dec 24 2021” and a birthday of “January 1 2000,” it should return a time difference of 8 days until the user’s next birthday.
  • Inputs:
    • birthday: A string in month-day-year format giving the user’s birthday
    • from_when: A string in month-day-year format. The number of days from this day to the user’s upcoming birthday will be printed for the user. Set the default value for this argument to be today().
  • Notes:
    • Your function should be able to handle all of the following date formats: “Jan 1 2000”, “January 1 2000”, “Jan 1, 2000”, “January 1, 2000”, “1/1/2000”, “1-1-2000”.
    • The two dates could be in different formats.
    • The lubridate cheat sheet will be helpful.
    • You can subtract date objects with -.
Code
# Write your function here

days_to_birthday <- function(birthday, from_when = today()) {
  parse_flex <- function(d) {
    as_date(parse_date_time(
      as.character(d), 
      orders = c("ymd", "mdy", "Bdy"),
      quiet = TRUE)
      )
    }

  bday <- parse_flex(birthday)
  from <- parse_flex(as.character(from_when))

  bday_this_year <- make_date(year(from), month(bday), day(bday))
  bday_next <- if (bday_this_year > from) bday_this_year else
               make_date(year(from) + 1, 
                         month(bday), day(bday))

  bday_next-from
}

course_names_to_df <- function(courses, separate_cross_list = TRUE) {
  dept_f <- str_extract(courses,"^[A-Z]+(/[A-Z]+)?")
  course_num <- as.numeric(str_extract(courses, "\\d+"))
  course_name <- str_extract(courses, "(?<=:\\s).+$")
  
  if (separate_cross_list) { 
    tibble(
      dept1 = str_extract(dept_f, "^[A-Z]+"),
      dept2 = str_extract(dept_f, "(?<=/)[A-Z]+"),
      course_num, course_name
    )
  } else {
    tibble(dept = dept_f, 
           course_num, course_name)
  }
}

Ex 2

Write code that reads in all the JSON files in the objects folder and returns a tibble with the following columns: ‘id’,‘accession_number’,‘title’,‘artist’,‘classification’,‘continent’,‘country’,‘dated’,‘life_date’,‘department’,‘culture’,‘nationality’,‘room’,‘style’,‘medium’,‘dimension’,‘description’,‘object_name’

Code
library(tidyverse)
library(here)
library(jsonlite)

path <- here("hw07/data/raw/collection-main/objects")

files <- dir(path, pattern = "*.json",recursive = TRUE)
file_paths <- file.path(path,files)


nullToNA <- function(x){
  if(is.null(x)){
    x <- NA
  }
  x
}

read_json_file <- function(file_path){
 result <- try(jsonlite::read_json(file_path), silent = TRUE)
  if(inherits(result, "try-error")) return(NULL)
  
  cols <- c("id","accession_number","title","artist","classification",
            "continent","country","dated","life_date","department",
            "culture","nationality","room","style","medium","dimension",
            "description","object_name"
            )
  
  row <- lapply(cols, function(col) {
    tryCatch(nullToNA(result[[col]]), 
             error = function(e) NA)
    })
  names(row) <- cols
  row <- lapply(row, as.character)
  as.data.frame(row, stringsAsFactors = FALSE)
}

mia_collection <- map(file_paths, read_json_file) |> list_rbind()

Ex 4

Explore the continent variable. Create a “clean” version of the variable called continent_clean such that each object has 1 value from the following: “Europe”, “Africa”, “North America”, “Asia”, “South America”, “Antarctica”, “Oceania”, “Unknown Continent”. As you clean the variable, state your assumptions explicitly so that someone could replicate your work based on your words. Remember to read the data your stored in the previous step first.

Assumptions: Records with a missing continent or non-match would be assigned as “Unknown” to keep all of the records rather than dropping them. Those containing Central America”, “Caribbean”, or “Mesoamerica” would be assigned as “North America” since those regions are of that continent. “Australia” would be assigned to “Oceania” since it is geographically in that region and “Middle East” would be mapped to “Asia” since those areas are within the Asian continent geographically

Code
mia_collection <- read_csv(here("hw07/data/processed/mia_collection.csv"))

mia_collection <- mia_collection |>
  mutate(continent_clean = case_when(
    str_detect(continent, regex("europe", ignore_case=TRUE)) ~ "Europe",
    str_detect(continent, regex("africa", ignore_case=TRUE)) ~ "Africa",
    str_detect(continent, regex("north america|central america|caribbean|mesoamerica|mexico", ignore_case = TRUE)) ~ "North America",
    str_detect(continent, regex("south america", ignore_case=TRUE)) ~ "South America",
    str_detect(continent, regex("asia|middle east|near east|far east|southeast asia|east asia", ignore_case = TRUE)) ~ "Asia",
    str_detect(continent, regex("oceania|pacific islands|australia", ignore_case = TRUE)) ~ "Oceania",
    str_detect(continent, regex("antarctica", ignore_case = TRUE)) ~ "Antarctica",
    
    is.na(continent) | str_trim(continent) == "" ~ "Unknown Continent", TRUE ~ "Unknown Continent"
  ))

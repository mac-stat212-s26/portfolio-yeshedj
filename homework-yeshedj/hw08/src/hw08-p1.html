Ex 3

Obtain the following information for all cheeses in the alphabetical database https://www.cheese.com/alphabetical/:

  • Cheese name
  • URL for the cheese’s webpage (e.g., https://www.cheese.com/gouda/)
  • Whether or not the cheese has a picture (e.g., gouda has a picture, but bianco does not)

To be kind to the website owners, please add a 1 second pause between page queries. (Note that you can view 100 cheeses at a time.)

Code
# Load required libraries
library(rvest)
library(dplyr)


# Define the URL
base_url    <- "https://www.cheese.com/alphabetical/?page="
scrape_page <- function(page_num) {
    url <- paste0(base_url, page_num)
  
# Read the HTML content from the webpage
  webpage <- read_html(url)
  
# Extract the cheese names and URLs
  h3_links <- webpage |> html_elements("h3 a")
  names    <- h3_links |> html_text(trim = TRUE)
  urls     <- h3_links |> html_attr("href") |>
    (\(x) paste0("https://www.cheese.com",x))()
  
# Check if the cheese has a picture
  all_srcs    <- webpage|> html_elements("img") |> html_attr("src")
  cheese_srcs <- all_srcs[grepl("cheese-thumbs|icon-cheese-default", all_srcs)]
  has_image   <- !grepl("icon-cheese-default", cheese_srcs)
  Sys.sleep(1)
  
# Create a data frame to store the results
  tibble(name = names, url = urls, has_image = has_image)
}
cheese_df <- bind_rows(lapply(1:103, scrape_page))

Ex 4

When you go to a particular cheese’s page (like gouda), you’ll see more detailed information about the cheese. For just 10 of the cheeses in the database, obtain the following detailed information: milk information, country of origin, family, type, and flavour. (Just 10 to avoid overtaxing the website. Continue adding a 1 second pause between page queries.)

Code
get_field <- function(items, label) {
  hit <- items[grepl(label, items, ignore.case = TRUE)]
  if (length(hit) == 0) return(NA_character_)
  trimws(gsub(paste0("^.*", label, "\\s*:?\\s*"), "", hit[1], ignore.case = TRUE))
}

scrape_detail <- function(cheese_url) {
  page<- read_html(cheese_url)
  items <- page |>
    html_elements("#collapse-information li") |>
    html_text(trim = TRUE)
  
  milk <-get_field(items, "Made from")
  country <- get_field(items, "Country of origin")
  family <- get_field(items, "Family")
  type <- get_field(items, "Type")
  flavor <- get_field(items, "Flavou?r")   # so that it matches Flavour or Flavor
  
  Sys.sleep(1)
  
  tibble(url = cheese_url, milk, country, family, type, flavor)
}

cheese_details <- bind_rows(lapply(cheese_df$url[1:10], scrape_detail))
